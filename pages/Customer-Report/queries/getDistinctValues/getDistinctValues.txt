WITH cpv_one AS (
  SELECT customer_id, code,
         MIN(name) AS name,
         MIN(remittance_name) AS remittance_name,
         MIN(main_phone) AS main_phone,
         (ARRAY_AGG(remittance_address))[1] AS remittance_address
  FROM bill_management_v2.customers_providers_vendors
  WHERE customer_id = {{ ReportSpecs.customerIdSql() }}
  GROUP BY customer_id, code
),
pv_one AS (
  SELECT code,
         CASE WHEN COUNT(DISTINCT btrim(name)) = 1 THEN MIN(name) END AS name,
         CASE WHEN COUNT(DISTINCT btrim(remittance_name)) = 1 THEN MIN(remittance_name) END AS remittance_name,
         CASE WHEN COUNT(DISTINCT btrim(main_phone)) = 1 THEN MIN(main_phone) END AS main_phone,
         CASE WHEN COUNT(*) = 1 THEN (ARRAY_AGG(remittance_address))[1] END AS remittance_address
  FROM bill_management_v2.providers_vendors
  WHERE code IN (SELECT vendor_code FROM bill_management_v2.analytics_monthly_feed
                 WHERE customer_id = {{ ReportSpecs.customerIdSql() }})
  GROUP BY code
),
vas_one AS (
  SELECT DISTINCT ON (s.virtual_account_id)
         s.virtual_account_id, s.account_status, s.account_opened, s.account_closed
  FROM bill_management_v2.virtual_accounts_status s
  JOIN bill_management_v2.virtual_accounts sva ON sva.id = s.virtual_account_id
  WHERE sva.customer_id = {{ ReportSpecs.customerIdSql() }}
  ORDER BY s.virtual_account_id, s.account_closed DESC NULLS LAST
),
rv_one AS (
  SELECT customer_id, vendor_code,
         NULLIF(btrim(MIN(providers_vendors_attributes::jsonb ->> 'Vendor Code')), '') AS fiq_vendor_id
  FROM bill_management_v2.reports_vendors
  WHERE customer_id = {{ ReportSpecs.customerIdSql() }}
  GROUP BY customer_id, vendor_code
),
attr_vals AS (
  SELECT vam.virtual_account_id,
         vmeta.attribute_name,
         string_agg(DISTINCT vam.attribute_value, ' | ' ORDER BY vam.attribute_value) AS val
  FROM bill_management_v2.virtual_accounts_attributes_mapping vam
  JOIN bill_management_v2.virtual_accounts_attributes_metadata vmeta
    ON vmeta.id = vam.virtual_accounts_attributes_metadata_id
  WHERE vmeta.customer_id = {{ ReportSpecs.customerIdSql() }}
    AND vmeta.deleted_at IS NULL
  GROUP BY vam.virtual_account_id, vmeta.attribute_name
),
feed_scoped AS (
  {{ ReportSpecs.feedCte() }}
),
amf_first AS (
  SELECT va2.account_code, f.location_id, f.vendor_code,
         MIN(f.time_period) AS first_period,
         MAX(f.statement_date) AS last_bill_date
  FROM bill_management_v2.analytics_monthly_feed f
  JOIN bill_management_v2.virtual_accounts va2 ON va2.id = f.virtual_account_id
  WHERE f.customer_id = {{ ReportSpecs.customerIdSql() }}
  GROUP BY va2.account_code, f.location_id, f.vendor_code
),
bill_history AS (
  SELECT ah.bill_id,
         ah.virtual_account_id,
         upper(btrim(ah.commodity)) AS commodity,
         MIN(ah.start_date)  AS start_date,
         MAX(ah.end_date)    AS end_date,
         SUM(ah.subcharges)  AS subcharges,
         SUM(ah.consumption) AS consumption
  FROM bill_management_v2.account_history ah
  JOIN bill_management_v2.virtual_accounts bva ON bva.id = ah.virtual_account_id
  WHERE bva.customer_id = {{ ReportSpecs.customerIdSql() }}
  GROUP BY ah.bill_id, ah.virtual_account_id, upper(btrim(ah.commodity))
),
gl_rows AS (
  SELECT vam.virtual_account_id,
         (substring(vmeta.attribute_name from '([0-9]+)'))::int AS gl_slot,
         MAX(NULLIF(btrim(vam.attribute_value), '')) FILTER (WHERE vmeta.attribute_name ~* '^gl\s*code')  AS gl_code,
         MAX(NULLIF(btrim(vam.attribute_value), '')) FILTER (WHERE vmeta.attribute_name ~* '^gl\s*alloc') AS gl_allocation
  FROM bill_management_v2.virtual_accounts_attributes_mapping vam
  JOIN bill_management_v2.virtual_accounts_attributes_metadata vmeta
    ON vmeta.id = vam.virtual_accounts_attributes_metadata_id
  WHERE vmeta.customer_id = {{ ReportSpecs.customerIdSql() }}
    AND vmeta.deleted_at IS NULL
    AND vmeta.attribute_name ~* '^gl\s*(code|alloc)'
  GROUP BY 1, 2
),
lf_bills AS (
  SELECT f.virtual_account_id, f.bill_id,
         MIN(f.bill_record_id)       AS bill_record_id,
         MIN(f.statement_date)::date AS bill_date,
         SUM(f.total_charges)        AS bill_amount
  FROM bill_management_v2.analytics_monthly_feed f
  WHERE f.customer_id = {{ ReportSpecs.customerIdSql() }}
  GROUP BY f.virtual_account_id, f.bill_id
),
lf_fee AS (
  SELECT li.virtual_account_id, li.bill_id,
         SUM(CASE WHEN li.charge > 0 THEN li.charge ELSE 0 END) AS late_fee_charged,
         SUM(CASE WHEN li.charge < 0 THEN li.charge ELSE 0 END) AS late_fee_recouped,
         SUM(li.charge)                                         AS late_fee
  FROM bill_management_v2.analytics_billing_line_items li
  WHERE li.customer_id = {{ ReportSpecs.customerIdSql() }}
    AND li.code = 'LATEFEE'
  GROUP BY li.virtual_account_id, li.bill_id
),
lf_seq AS (
  SELECT b.virtual_account_id, b.bill_id, b.bill_date, b.bill_amount,
         fee.late_fee, fee.late_fee_charged, fee.late_fee_recouped,
         LAG(b.bill_date)      OVER w AS prev_bill_date,
         LAG(b.bill_amount)    OVER w AS prev_bill_amount,
         LAG(b.bill_record_id) OVER w AS prev_bill_record_id
  FROM lf_bills b
  LEFT JOIN lf_fee fee ON fee.virtual_account_id = b.virtual_account_id AND fee.bill_id = b.bill_id
  WINDOW w AS (PARTITION BY b.virtual_account_id ORDER BY b.bill_date, b.bill_id)
)
SELECT DISTINCT {{ ReportSpecs.distinctExpr() }} AS value
FROM {{ ReportSpecs.distinctFrom() }}
{{ ReportSpecs.distinctWhere() }}
  AND {{ ReportSpecs.distinctExpr() }} IS NOT NULL
ORDER BY 1
LIMIT {{ ReportSpecs.distinctLimit() }}
OFFSET {{ ReportSpecs.distinctOffset() }};
